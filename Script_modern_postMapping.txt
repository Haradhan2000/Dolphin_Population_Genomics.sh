##################################################
### Convert SAM to sorted BAM
############################################################

for file in 04_mapping/*.sam
do
    sample=$(basename "$file" .sam)

    echo "Processing ${sample}..."

    samtools view -@ 32 -b "$file" | \
    samtools sort -@ 32 \
        -o 05_bam_files/${sample}.sorted.bam

    echo "${sample} completed."
done

echo "All SAM files converted to sorted BAM successfully."
#################################################################################
### Indexing BAM
############################################################
for file in 05_bam_files/*.sorted.bam
do
    echo "Indexing $(basename "$file")..."
    samtools index -@ 32 "$file"
done

echo "All BAM files indexed successfully."
############################################################
### Mapping statistics using Qualimap
############################################################

for file in 05_bam_files/*.sorted.bam
do
    sample=$(basename "$file" .sorted.bam)

    echo "Running Qualimap for ${sample}..."

    qualimap bamqc \
        -bam "$file" \
        -outdir 06_mapping_statistics/${sample}_qualimap \
        -nt 32

    echo "${sample} completed."
done

echo "Qualimap analysis completed."
#####################################################################
#### Mark duplicates
#################################################################

for file in 05_bam_files/*.sorted.bam
do
    sample=$(basename "$file" .sorted.bam)

    gatk MarkDuplicates \
        -I "$file" \
        -O 09_markduplicates/${sample}.markdup.bam \
        -M 09_markduplicates/${sample}_duplication_metrics.txt \
        --REMOVE_DUPLICATES true
done

echo "Duplicate removal completed."
#####################################################################
####### Indexing deduplicated BAM
for bam in 09_markduplicates/*_deduplicated.bam
do
    samtools index -@ 32 "$bam"
done

echo "All deduplicated BAM files indexed successfully."
##############################################################
### Calling VCF (variant call format) files
#########################################################################
configureStrelkaGermlineWorkflow.py \
    --bam 09_markduplicates/SRR11431927_deduplicated.bam \
    --bam 09_markduplicates/SRR13005646_deduplicated.bam \
    --bam 09_markduplicates/SRR33006009_deduplicated.bam \
    --bam 09_markduplicates/SRR39498552_deduplicated.bam \
    --referenceFasta ~/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa \
    --runDir 10_variants/strelka_output

### Run
10_variants/strelka_output/runWorkflow.py -m local -j 32
