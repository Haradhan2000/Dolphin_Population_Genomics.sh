#!/bin/bash

MODERN_DIR="$HOME/compre_result_paleomix/02_modern"
OUT_DIR="$HOME/compre_result_paleomix/05_qualimap_results"

mkdir -p "$OUT_DIR"

for bam in \
    "$MODERN_DIR/SRR11431927.dolphin_chr.bam" \
    "$MODERN_DIR/SRR13005646.dolphin_chr.bam" \
    "$MODERN_DIR/SRR39498552.dolphin_chr.bam"
do
    if [ ! -f "$bam" ]; then
        echo "BAM not found: $bam"
        continue
    fi

    filename=$(basename "$bam" .bam)
    prefix=${filename%%.*}
    sample_out="$OUT_DIR/bamqc_${prefix}"

    if [ -f "$sample_out/genome_results.txt" ]; then
        echo "Skipping $prefix — Qualimap already completed."
        continue
    fi

    mkdir -p "$sample_out"

    echo "Running Qualimap for $prefix..."

    qualimap bamqc \
        -bam "$bam" \
        -outdir "$sample_out" \
        -outformat PDF \
        -nt 16

    echo "$prefix completed."
done

echo "Qualimap analysis completed."
