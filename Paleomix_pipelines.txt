#!/bin/bash

MODERN_DIR="$HOME/compre_result_paleomix/02_modern"
OUT_DIR="$HOME/compre_result_paleomix/05_qualimap_results"

mkdir -p "$OUT_DIR"

for bam in \
    "$MODERN_DIR/SRR11431927.dolphin_chr.bam" \
    "$MODERN_DIR/SRR13005646.dolphin_chr.bam" \
    "$MODERN_DIR/SRR39498552.dolphin_chr.bam"
do
    if [ ! -f "$bam" ]; then
        echo "BAM not found: $bam"
        continue
    fi

    filename=$(basename "$bam" .bam)
    prefix=${filename%%.*}
    sample_out="$OUT_DIR/bamqc_${prefix}"

    if [ -f "$sample_out/genome_results.txt" ]; then
        echo "Skipping $prefix — Qualimap already completed."
        continue
    fi

    mkdir -p "$sample_out"

    echo "Running Qualimap for $prefix..."

    qualimap bamqc \
        -bam "$bam" \
        -outdir "$sample_out" \
        -outformat PDF \
        -nt 16

    echo "$prefix completed."
done

echo "Qualimap analysis completed."
#########################################################################################
##################################################################################################
# Modern Dolphin Nucleotide Diversity Analysis Using ANGSD
# Software: ANGSD, realSFS
# Reference: RaGOO-improved dolphin genome
# Samples: SRR11431927, SRR13005646
Step 1. Generate the site allele frequency likelihood (SAF) files

REF=/home/haradhan/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa

for sample in SRR11431927 SRR13005646
do
    angsd \
        -bam 06_nucleotide_diversity/${sample}/${sample}.list \
        -ref "$REF" \
        -anc "$REF" \
        -out 06_nucleotide_diversity/${sample}/${sample} \
        -doSaf 1 \
        -GL 1 \
        -minMapQ 30 \
        -minQ 20 \
        -remove_bads 1 \
        -uniqueOnly 1 \
        -doCounts 1 \
        -P 8
done
###############################################################################
Step 2. Estimate the site frequency spectrum (SFS)

for sample in SRR11431927 SRR13005646
do
    realSFS \
        06_nucleotide_diversity/${sample}/${sample}.saf.idx \
        > 06_nucleotide_diversity/${sample}/${sample}.sfs
done
#################################################################################
for sample in SRR11431927 SRR13005646
do
    echo "Calculating theta for $sample"

    realSFS saf2theta \
        06_nucleotide_diversity/${sample}/${sample}.saf.idx \
        -sfs 06_nucleotide_diversity/${sample}/${sample}.sfs \
        -outname 06_nucleotide_diversity/${sample}/${sample}

    thetaStat do_stat \
        06_nucleotide_diversity/${sample}/${sample}.thetas.idx

    echo "$sample theta calculation completed"
done
###################################################################################
for sample in SRR11431927 SRR13005646
do
    echo "======================================"
    echo "Calculating 50-kb window statistics for $sample"
    echo "======================================"

    thetaStat do_stat \
        06_nucleotide_diversity/${sample}/${sample}.thetas.idx \
        -win 50000 \
        -step 50000 \
        -outnames 06_nucleotide_diversity/${sample}/${sample}_window

    echo "$sample window statistics completed"
done
########################################################################################
for sample in SRR11431927 SRR13005646
do
    echo "===== $sample ====="

    awk '
    BEGIN { FS="[ \t]+" }
    /^#/ { next }
    NF >= 16 &&
    $7 ~ /^[0-9]+([.][0-9]+)?([eE][-+]?[0-9]+)?$/ &&
    $16 ~ /^[0-9]+$/ && $16 > 0 {
        sumTP += $7
        sumSites += $16
        nWindows++
    }
    END {
        if (sumSites > 0) {
            printf "Valid windows: %d\n", nWindows
            printf "Sum of tP: %.6f\n", sumTP
            printf "Sum of nSites: %.0f\n", sumSites
            printf "Preliminary pi per site: %.10f\n", sumTP/sumSites
        } else {
            print "No valid windows found"
        }
    }' "06_nucleotide_diversity/${sample}/${sample}_window.pestPG"
done
