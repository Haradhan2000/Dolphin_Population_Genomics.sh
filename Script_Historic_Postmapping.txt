############################################
#1 Generating SAM files 
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do
    echo "Generating SAM for ${sample}..."

    bwa samse \
        /home/haradhan/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa \
        04_mapping/${sample}.sai \
        03_trimmed/${sample}.trimmed_combined.fastq.gz \
        > 04_mapping/${sample}.sam

done
############################################
#2 Convert SAM to sorted BAM
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Sorting ${sample}..."

    samtools sort \
        -O bam \
        04_mapping/${sample}.sam \
        -o 05_bam_files/${sample}.bam

done
############################################
#3 # 3. Index sorted BAM files
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Indexing filtered ${sample}..."

samtools index \
05_bam_files/${sample}.filtered.bam

done
############################################
# 4. Qualimap QC of raw sorted BAM files
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Running Qualimap BAMQC for ${sample}..."

    qualimap bamqc \
        -bam 05_bam_files/${sample}.bam \
        -nt 2 \
        -outdir qualimap_results_raw/${sample} \
        -outformat HTML

done

################################################################
############################################
# 5. Filter BAM files
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Filtering ${sample}..."

    samtools view \
        -h \
        -F 4 \
        -q 30 \
        -b \
        05_bam_files/${sample}.bam \
        > 05_bam_files/${sample}.filtered.bam

done
############################################
# 6. Index filtered BAM files
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Indexing filtered ${sample}..."

    samtools index \
        05_bam_files/${sample}.filtered.bam

done
############################################
############################################
# 7. Remove PCR duplicates
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Removing duplicates for ${sample}..."

    samtools markdup \
        -r \
        -s \
        05_bam_files/${sample}.filtered.bam \
        05_bam_files/${sample}.filtered.dedup.bam

done
############################################
############################################
# 8. Index deduplicated BAM files
############################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Indexing deduplicated ${sample}..."

    samtools index \
        05_bam_files/${sample}.filtered.dedup.bam

done
#################################################
#9.Assessment of DNA damage using DamageProfiler
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Running DamageProfiler for ${sample}..."

    mkdir -p 09_damage_profiler/${sample}

    damageprofiler \
        -i 05_bam_files/${sample}.filtered.dedup.bam \
        -r /home/haradhan/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa \
        -t 2 \
        -o 09_damage_profiler/${sample} \
        -yaxis_dp_max 0.30

done
######################################################
############################################
# 10. Assess BAM quality using Qualimap
############################################

mkdir -p qualimap_results

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

    echo "Running Qualimap BAMQC for ${sample}..."

    qualimap bamqc \
        -bam 05_bam_files/${sample}.filtered.dedup.bam \
        -nt 2 \
        -outdir qualimap_results/${sample} \
        -outformat HTML

done
