############################################
#1 Convert SAM to sorted BAM
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Sorting ${sample}"

samtools sort \
-@ 8 \
-O bam \
04_mapping/${sample}.sam \
-o 05_bam_files/${sample}.sorted.bam

done
############################################
#2 Mapping statistics using Qualimap
############################################

for file in 05_bam_files/*.sorted.bam
do
    sample=$(basename "$file" .sorted.bam)

    echo "Running Qualimap for ${sample}..."

    qualimap bamqc \
        --java-mem-size=16G \
        -bam "$file" \
        -outdir 06_mapping_statistics/${sample}_qualimap \
        -nt 16

    echo "${sample} completed."
done

echo "Qualimap analysis completed."
##################################################################
#3 Filter mapped reads
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Filtering ${sample}..."

samtools view \
-h \
-F 4 \
-q 30 \
-b \
05_bam_files/${sample}.sorted.bam \
> 05_bam_files/${sample}.filtered.bam

done
############################################
#4 Index filtered BAM files
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Indexing filtered ${sample}..."

samtools index \
05_bam_files/${sample}.filtered.bam

done

################################################################
#5 Mapping statistics using Qualimap on filtered BAM files
############################################

for file in 05_bam_files/*.filtered.bam
do
    sample=$(basename "$file" .filtered.bam)

    echo "Running Qualimap for ${sample}..."

    qualimap bamqc \
        --java-mem-size=16G \
        -bam "$file" \
        -outdir 06_mapping_statistics/${sample}_filtered_qualimap \
        -nt 16

    echo "${sample} completed."
done

echo "Qualimap analysis completed."
########################################################################
#6 Name-sort filtered BAM files for fixmate processing
##################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Name sorting ${sample}"

samtools sort \
-n \
-@ 8 \
-O bam \
05_bam_files/${sample}.filtered.bam \
-o 05_bam_files/name_sorted_bam/${sample}.namesorted.bam

done

############################################
#7 Run fixmate
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Running fixmate for ${sample}"

samtools fixmate \
-m \
05_bam_files/name_sorted_bam/${sample}.namesorted.bam \
05_bam_files/fixmate_bam/${sample}.fixmate.bam

done
############################################
#8 Coordinate sort fixmate BAM files
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Coordinate sorting ${sample}"

samtools sort \
-@ 8 \
-O bam \
05_bam_files/fixmate_bam/${sample}.fixmate.bam \
-o 05_bam_files/fixmate_sorted_bam/${sample}.fixmate.sorted.bam

done
############################################
#9 Remove PCR duplicates
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Removing duplicates from ${sample}"

samtools markdup \
-r \
-s \
05_bam_files/fixmate_sorted_bam/${sample}.fixmate.sorted.bam \
09_markduplicates/${sample}.dedup.bam

done

############################################
#10 Index deduplicated BAM files
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Indexing deduplicated ${sample}..."

samtools index \
09_markduplicates/${sample}.dedup.bam

done
