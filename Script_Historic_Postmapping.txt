############################################
# Convert SAM to sorted BAM
############################################

for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Sorting ${sample}"

samtools sort \
-@ 8 \
-O bam \
04_mapping/${sample}.sam \
-o 05_bam_files/${sample}.sorted.bam

done
