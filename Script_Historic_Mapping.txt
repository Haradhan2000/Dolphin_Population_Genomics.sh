#################################################
#1 Check FASTQ file sizes
ls -lh *.fastq.gz
##################################################
Data processing and pre-mapping
#################################################
#2 QC chceck and generate multiQC
###########################################################################
fastqc 01_raw_data/*.fastq.gz \
    -o 02_fastqc_reports

cd 02_fastqc_reports

multiqc .
###########################################################################
#3 Adapter removal_trimming
############################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

AdapterRemoval \
--file1 01_raw_data/${sample}_*_R1_001.fastq.gz \
--file2 01_raw_data/${sample}_*_R2_001.fastq.gz \
--basename 03_trimmed/${sample} \
--gzip \
--threads 8 \
--qualitymax 41 \
--collapse \
--trimns \
--trimqualities \
--adapter1 AGATCGGAAGAGCACACGTCTGAACTCCAGTCAC \
--adapter2 AGATCGGAAGAGCGTCGTGTAGGGAAAGAGTGTA \
--minlength 30 \
--minquality 20 \
--minadapteroverlap 1

done
######################################################################
#4 QC chceck and generate multiQC of trimmed files 
######################################################################
fastqc 03_trimmed/*.pair1.truncated.gz \
       03_trimmed/*.pair2.truncated.gz \
       03_trimmed/*.collapsed.gz \
       03_trimmed/*.collapsed.truncated.gz \
       -o 03_trimmed/fastqc_report

#####################################################################
#5 Combine AdapterRemoval output files into a single trimmed FASTQ per sample
#####################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Combining files for ${sample}"

cat ${sample}.collapsed.gz \
    ${sample}.collapsed.truncated.gz \
    ${sample}.singleton.truncated.gz \
    ${sample}.pair1.truncated.gz \
    ${sample}.pair2.truncated.gz \
    > ${sample}.trimmed_combined.fastq.gz

done

ls -lh *.trimmed_combined.fastq.gz
###############################################################################################################################################
#6 Data Mapping (Reference genome: 07_indexes/dolphin_chr.fa)
###############################################################################################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Mapping ${sample}"

bwa mem \
-t 8 \
07_indexes/dolphin_chr.fa \
03_trimmed/${sample}.pair1.truncated.gz \
03_trimmed/${sample}.pair2.truncated.gz \
> 04_mapping/${sample}.sam

done
