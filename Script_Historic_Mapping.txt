ls -lh *.fastq.gz
##################################################
Data processing and pre-mapping
#################################################
# QC chceck and generate multiQC
###########################################################################
fastqc 01_raw_data/*.fastq.gz \
    -o 02_fastqc_reports

cd 02_fastqc_reports

multiqc .
###########################################################################
# Adapter removal_trimming
############################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

AdapterRemoval \
--file1 01_raw_data/${sample}_*_R1_001.fastq.gz \
--file2 01_raw_data/${sample}_*_R2_001.fastq.gz \
--basename 03_trimmed/${sample} \
--gzip \
--threads 8 \
--qualitymax 41 \
--collapse \
--trimns \
--trimqualities \
--adapter1 AGATCGGAAGAGCACACGTCTGAACTCCAGTCAC \
--adapter2 AGATCGGAAGAGCGTCGTGTAGGGAAAGAGTGTA \
--minlength 30 \
--minquality 20 \
--minadapteroverlap 1

done
######################################################################
# QC chceck and generate multiQC of trimmed files 
######################################################################
fastqc 03_trimmed/*.gz \
-o 02_fastqc_reports
cd 02_fastqc_reports
multiqc .
#####################################################################
# Combine AdapterRemoval output files into a single trimmed FASTQ per sample
#####################################################################
for sample in DP_1 DP_2 DP_4 DP_6 DP_7 DP_8 DP_9 DP_12 DP_14
do

echo "Combining files for ${sample}"

cat ${sample}.collapsed.gz \
    ${sample}.collapsed.truncated.gz \
    ${sample}.singleton.truncated.gz \
    ${sample}.pair1.truncated.gz \
    ${sample}.pair2.truncated.gz \
    > ${sample}.trimmed_combined.fastq.gz

done

ls -lh *.trimmed_combined.fastq.gz
###############################################################################################################################################

Data Mapping
###################################################################################################################################################################################
# Activate environment
############################################

conda activate dolphin_wgs


############################################
# Install mapping tools (run once)
############################################

conda install -c bioconda bwa samtools -y


############################################
# Create output folders
############################################

mkdir -p ~/Dolphin_WGS_Analysis/analysis_output/{mapped,sorted_bam,dedup_bam}


############################################
# Move to reference genome folder (NA)
############################################

cd ~/Dolphin_WGS_Analysis/reference_genome

# Check reference genome file name
ls


############################################
# Index reference genome (run once only) (NA)
############################################

bwa index dolphin_reference.fasta

samtools faidx dolphin_reference.fasta


############################################
# Move to project directory
############################################

cd ~/Dolphin_WGS_Analysis


for sample in DP_1 DP_2
do

echo "Mapping ${sample}"

bwa mem \
-t 8 \
07_indexes/dolphin_chr.fa \
analysis_output/trimmed/${sample}.pair1.truncated.gz \
analysis_output/trimmed/${sample}.pair2.truncated.gz \
> analysis_output/mapped/${sample}.sam

done


############################################
# Index sorted BAM files
############################################

for sample in DP_1 DP_2
do

echo "Sorting ${sample}"

samtools sort \
-@ 8 \
-O bam \
analysis_output/mapped/${sample}.sam \
-o analysis_output/sorted_bam/${sample}.sorted.bam

done
############################################
# Filter mapped reads
############################################

for sample in DP_1 DP_2
do

echo "Filtering ${sample}..."

samtools view \
-h \
-F 4 \
-q 30 \
-b \
analysis_output/sorted_bam/${sample}.sorted.bam \
> analysis_output/sorted_bam/${sample}.filtered.bam

done


############################################
# Index filtered BAM files
############################################

for sample in DP_1 DP_2
do

echo "Indexing filtered ${sample}..."

samtools index \
analysis_output/sorted_bam/${sample}.filtered.bam

done

############################################
# Create directories
############################################

mkdir -p analysis_output/name_sorted_bam
mkdir -p analysis_output/fixmate_bam
mkdir -p analysis_output/fixmate_sorted_bam


############################################
# Name-sort BAM files
############################################

for sample in DP_1 DP_2
do

echo "Name sorting ${sample}"

samtools sort \
-n \
-@ 8 \
-O bam \
analysis_output/sorted_bam/${sample}.filtered.bam \
-o analysis_output/name_sorted_bam/${sample}.namesorted.bam

done


############################################
# Run fixmate
############################################

for sample in DP_1 DP_2
do

echo "Running fixmate for ${sample}"

samtools fixmate \
-m \
analysis_output/name_sorted_bam/${sample}.namesorted.bam \
analysis_output/fixmate_bam/${sample}.fixmate.bam

done


############################################
# Coordinate sort fixmate BAM files
############################################

for sample in DP_1 DP_2
do

echo "Sorting fixmate BAM for ${sample}"

samtools sort \
-@ 8 \
-O bam \
analysis_output/fixmate_bam/${sample}.fixmate.bam \
-o analysis_output/fixmate_sorted_bam/${sample}.fixmate.sorted.bam

done

# Remove PCR duplicates
############################################

for sample in DP_1 DP_2
do

echo "Removing duplicates from ${sample}"

samtools markdup \
-r \
-s \
analysis_output/fixmate_sorted_bam/${sample}.fixmate.sorted.bam \
analysis_output/dedup_bam/${sample}.dedup.bam

done

############################################
# Index deduplicated BAM files
############################################

for sample in DP_1 DP_2
do

echo "Indexing deduplicated ${sample}..."

samtools index \
analysis_output/dedup_bam/${sample}.dedup.bam

done


############################################
# Generate mapping statistics
############################################

for sample in DP_1 DP_2
do

echo "Generating flagstat for ${sample}..."

samtools flagstat \
analysis_output/dedup_bam/${sample}.dedup.bam \
> analysis_output/dedup_bam/${sample}.flagstat.txt

done


############################################
# Check output files
############################################

ls analysis_output/dedup_bam
