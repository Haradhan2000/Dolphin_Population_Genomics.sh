#!/bin/bash
############################################################
# Modern Dolphin WGS Pipeline
############################################################
# Activate conda environment
############################
conda activate dolphin_wgs
## run fastqc
###########################################
conda activate fastqc

fastqc \
    01_raw_data/*.fastq \
    -t 16 \
    -o 02_fastqc_reports
multiqc \
    02_fastqc_reports \
    -o 02_fastqc_reports
#######################################
### Trimming 
############################################
conda activate dolphin_wgs

for file in 01_raw_data/*_1.fastq
do
    mate="${file/_1.fastq/_2.fastq}"

    trim_galore \
        --paired \
        --cores 32 \
        --output_dir 03_trimmed \
        "$file" "$mate"
done
###########################################################
###Fastqc reports after trimming
#######################################################
fastqc \
    03_trimmed/*_val_*.fq \
    -t 32 \
    -o 03_trimmed && \
multiqc \
    03_trimmed \
    -o 03_trimmed
####################################################
#### Mapping 
############################################################
for file1 in 03_trimmed/*_1_val_1.fq
do
    file2=${file1/_1_val_1.fq/_2_val_2.fq}
    sample=$(basename "$file1" _1_val_1.fq)

    echo "Mapping ${sample}..."

    bwa mem \
        -t 32 \
        ~/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa \
        "$file1" \
        "$file2" \
        > 04_mapping/${sample}.sam

    echo "${sample} completed."
done

echo "All samples have been mapped successfully."
#######################################################################
### Convert SAM to sorted BAM
############################################################

for file in 04_mapping/*.sam
do
    sample=$(basename "$file" .sam)

    echo "Processing ${sample}..."

    samtools view -@ 32 -b "$file" | \
    samtools sort -@ 32 \
        -o 05_bam_files/${sample}.sorted.bam

    echo "${sample} completed."
done

echo "All SAM files converted to sorted BAM successfully."
#################################################################################
### Indexing BAM
############################################################
for file in 05_bam_files/*.sorted.bam
do
    echo "Indexing $(basename "$file")..."
    samtools index -@ 32 "$file"
done

echo "All BAM files indexed successfully."
############################################################
### Mapping statistics using Qualimap
############################################################

for file in 05_bam_files/*.sorted.bam
do
    sample=$(basename "$file" .sorted.bam)

    echo "Running Qualimap for ${sample}..."

    qualimap bamqc \
        -bam "$file" \
        -outdir 06_mapping_statistics/${sample}_qualimap \
        -nt 32

    echo "${sample} completed."
done

echo "Qualimap analysis completed."
#####################################################################
