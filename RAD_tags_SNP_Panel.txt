Species: Orcaella brevirostris
File: NEW_ORCAELLA_RADs.fasta
#############################################
#1 Number of RAD loci/tags
grep -c "^>" 01_input/NEW_ORCAELLA_RADs.fasta
#################################################
#2 Check the length of each RAD-tag sequence
awk '
/^>/ {
    if (seq != "") print length(seq)
    seq=""
    next
}
{
    seq=seq $0
}
END {
    if (seq != "") print length(seq)
}' 01_input/NEW_ORCAELLA_RADs.fasta |
sort | uniq -c
###########################################################
# According to the collaborator SNP is located at the center of each RAD-tag sequence.
# For a 201-bp sequence:
# SNP position = (201 + 1) / 2 = 101
################################################################
03. Extract the central SNP nucleotide from each RAD tag
python -c 'from pathlib import Path; lines=Path("01_input/NEW_ORCAELLA_RADs.fasta").read_text().splitlines(); out=[]; i=0;
while i<len(lines):
    if lines[i].startswith(">"):
        out.append(lines[i][1:]+"|"+lines[i+1][100])
        i+=2
    else:
        i+=1
Path("02_analysis/RAD_tags_central_base.txt").write_text("\n".join(out)+"\n")'

# Number of RAD loci extracted
wc -l 02_analysis/RAD_tags_central_base.txt

# Distribution of the central nucleotide
cut -d'|' -f2 02_analysis/RAD_tags_central_base.txt | sort | uniq -c
########################################################################
### Ambiguous base (N)
awk '
/^>/ {
    if (seq != "") {
        if (seq ~ /N/) count++
    }
    seq=""
    next
}
{
    gsub(/\r/, "", $0)
    seq=seq $0
}
END {
    if (seq ~ /N/) count++
    print "RAD tags containing N:", count
}' 01_input/NEW_ORCAELLA_RADs.fasta
