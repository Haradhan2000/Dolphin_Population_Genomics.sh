--minlength 30 \
--minquality 20 \
--minadapteroverlap 1

done

###############################################################################################################################################################################
###############################################################################
# Dolphin chromosome-scale genome scaffolding pipeline
# Using RaGOO
###############################################################################

set -e
set -o pipefail

###############################################################################
# INPUT FILES
###############################################################################

QUERY="reference_genome/GCA_054084255.1_GACMC_Pgan_genomic.fna"

REFERENCE="reference_genome/GCF_011762595.2_mTurTru1.mat.Y_genomic.fna"

THREADS=20

###############################################################################
# CREATE OUTPUT DIRECTORIES
###############################################################################

mkdir -p \
01_QC_before \
02_filtered \
03_ragoo \
04_QC_after \
05_synteny \
06_final_reference \
07_indexes \
08_logs

###############################################################################
# STEP 1 — INITIAL QC
###############################################################################

echo "Running QUAST before scaffolding..."

quast.py \
${QUERY} \
-o 01_QC_before/quast_before \
-t ${THREADS} \
2>&1 | tee 08_logs/quast_before.log

echo "Running BUSCO before scaffolding..."

busco \
-i ${QUERY} \
-o busco_before \
-m genome \
-l mammalia_odb10 \
-c ${THREADS} \
--out_path 01_QC_before \
2>&1 | tee 08_logs/busco_before.log
###############################################################################
# STEP 2 — REMOVE SMALL SCAFFOLDS
###############################################################################

echo "Filtering scaffolds smaller than 5 kb..."

seqkit seq \
-m 5000 \
${QUERY} \
> 02_filtered/dolphin_filtered.fa

###############################################################################
# STEP 3 — RUN RAGOO
###############################################################################

ragoo.py \
02_filtered/dolphin_filtered.fa \
${REFERENCE} \
-t ${THREADS} \
-b \
2>&1 | tee 08_logs/ragoo.log
###############################################################################
# STEP 4 — FINAL ASSEMBLY
###############################################################################

cp \
03_ragoo/ragoo.fasta \
06_final_reference/dolphin_chr.fa

###############################################################################
# STEP 5 — POST-SCAFFOLDING QC
###############################################################################

echo "Running QUAST after scaffolding..."

THREADS=40

quast.py \
06_final_reference/dolphin_chr.fa \
-o 04_QC_after/quast_after \
-t ${THREADS} \
2>&1 | tee 08_logs/quast_after.log

echo "Running BUSCO after scaffolding..."

busco \
-i 06_final_reference/dolphin_chr.fa \
-o busco_after \
-m genome \
-l mammalia_odb10 \
-c ${THREADS} \
--out_path 04_QC_after \
2>&1 | tee 08_logs/busco_after.log

###############################################################################
# STEP 6 — SYNTENY VALIDATION
###############################################################################

echo "Running whole genome alignment validation..."

cd 05_synteny

nucmer \
../reference_genome/GCF_011762595.2_mTurTru1.mat.Y_genomic.fna \
../06_final_reference/dolphin_chr.fa \
-p dolphin

delta-filter \
-1 dolphin.delta \
> dolphin.filter.delta

mummerplot \
--fat \
--filter \
-p dolphin \
dolphin.filter.delta

cd ..
###############################################################################
# STEP 7 — COPY FINAL GENOME
###############################################################################

cp \
06_final_reference/dolphin_chr.fa \
07_indexes/

###############################################################################
# STEP 8 — BWA INDEX
###############################################################################

echo "Running BWA index..."

bwa index \
07_indexes/dolphin_chr.fa

###############################################################################
# STEP 9 — FASTA INDEX
###############################################################################

echo "Running samtools faidx..."

samtools faidx \
07_indexes/dolphin_chr.fa

###############################################################################
# STEP 10 — CREATE SEQUENCE DICTIONARY
###############################################################################

echo "Creating sequence dictionary..."

picard CreateSequenceDictionary \
R=07_indexes/dolphin_chr.fa \
O=07_indexes/dolphin_chr.dict

###############################################################################
# PIPELINE COMPLETE
###############################################################################

echo "PIPELINE COMPLETED SUCCESSFULLY"
echo "=================================================="

echo ""

echo "07_indexes/dolphin_chr.fa.bwt"

# Index sorted BAM files
############################################

do

echo "Sorting ${sample}"

samtools sort \
-@ 8 \
-O bam \
analysis_output/mapped/${sample}.sam \
-o analysis_output/sorted_bam/${sample}.sorted.bam

done
############################################
# Filter mapped reads
############################################

for sample in DP_1 DP_2
do

echo "Filtering ${sample}..."

samtools view \
-h \
-F 4 \
-q 30 \
-b \
analysis_output/sorted_bam/${sample}.sorted.bam \
> analysis_output/sorted_bam/${sample}.filtered.bam

done


############################################
# Index filtered BAM files
############################################

for sample in DP_1 DP_2
do

echo "Indexing filtered ${sample}..."

samtools index \
analysis_output/sorted_bam/${sample}.filtered.bam

done

############################################
# Create directories
############################################

mkdir -p analysis_output/name_sorted_bam
mkdir -p analysis_output/fixmate_bam
mkdir -p analysis_output/fixmate_sorted_bam


############################################
# Name-sort BAM files
############################################

for sample in DP_1 DP_2
do

echo "Name sorting ${sample}"

samtools sort \
-n \
-@ 8 \
-O bam \
analysis_output/sorted_bam/${sample}.filtered.bam \
-o analysis_output/name_sorted_bam/${sample}.namesorted.bam

done


############################################
# Run fixmate
############################################

for sample in DP_1 DP_2
do

echo "Running fixmate for ${sample}"

samtools fixmate \
-m \
analysis_output/name_sorted_bam/${sample}.namesorted.bam \
analysis_output/fixmate_bam/${sample}.fixmate.bam

done


############################################
# Coordinate sort fixmate BAM files
############################################

for sample in DP_1 DP_2
do

echo "Sorting fixmate BAM for ${sample}"

samtools sort \
-@ 8 \
-O bam \
analysis_output/fixmate_bam/${sample}.fixmate.bam \
-o analysis_output/fixmate_sorted_bam/${sample}.fixmate.sorted.bam

done

# Remove PCR duplicates
############################################

for sample in DP_1 DP_2
do

echo "Removing duplicates from ${sample}"

samtools markdup \
-r \
-s \
analysis_output/fixmate_sorted_bam/${sample}.fixmate.sorted.bam \
analysis_output/dedup_bam/${sample}.dedup.bam

done

############################################
# Index deduplicated BAM files
############################################

for sample in DP_1 DP_2
do

echo "Indexing deduplicated ${sample}..."

samtools index \
analysis_output/dedup_bam/${sample}.dedup.bam

done


############################################
# Generate mapping statistics
############################################

for sample in DP_1 DP_2
do

echo "Generating flagstat for ${sample}..."

samtools flagstat \
analysis_output/dedup_bam/${sample}.dedup.bam \
> analysis_output/dedup_bam/${sample}.flagstat.txt

done
