#!/bin/bash
############################################################
# Modern Dolphin WGS Pipeline
############################################################
# Activate conda environment
############################
conda activate dolphin_wgs
## run fastqc
###########################################
conda activate fastqc

fastqc \
    01_raw_data/*.fastq.gz \
    -t 16 \
    -o 02_fastqc_reports

multiqc \
    02_fastqc_reports \
    -o 02_fastqc_reports
#######################################
### Trimming 
############################################
for file in 01_raw_data/*_R1_001.fastq.gz
do
    mate="${file/_R1_001.fastq.gz/_R2_001.fastq.gz}"

    trim_galore \
        --paired \
        --cores 8 \
        --output_dir 03_trimmed \
        "$file" "$mate"
done
###########################################################
###Fastqc reports after trimming
#######################################################
fastqc \
    03_trimmed/*_val_*.fq.gz \
    -t 16 \
    -o 03_trimmed/fastqc_report && \
multiqc \
    03_trimmed/fastqc_report \
    -o 03_trimmed/fastqc_report \
    -n multiqc_trimming
####################################################
#### Mapping 
############################################################
for file1 in 03_trimmed/*_1_val_1.fq
do
    file2=${file1/_1_val_1.fq/_2_val_2.fq}
    sample=$(basename "$file1" _1_val_1.fq)

    echo "Mapping ${sample}..."

    bwa mem \
        -t 32 \
        -R "@RG\tID:${sample}\tSM:${sample}\tLB:lib1\tPL:ILLUMINA\tPU:unit1" \
        ~/Dolphin_WGS_Analysis/07_indexes/dolphin_chr.fa \
        "$file1" \
        "$file2" \
        > 04_mapping/${sample}.sam

    echo "${sample} completed."
done

echo "All samples have been mapped successfully."
#######################################################################
